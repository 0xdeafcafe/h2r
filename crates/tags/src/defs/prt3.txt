# prt3: particle. Layout from the schema and docs/formats/effects.md; field
# meanings E from the particle draw callback 0x00224550 (the prt3 accessor
# vtable 0x004535f0), docs/formats/effects.md "prt3 root". The property
# struct is effe's (defs/effe.txt), so this file comes after it.
group prt3 0xbc align 4
  0x0 u16 flags -- E 0x00224550: 0x2/0x4 random U/V mirror, 0x8 frames end on the last sprite, 0x10 sequence per particle, 0x20 no frame blend, 0x80/0x100 times the particle colour, 0x200 dies at rest, 0x2000 rotation +90 degrees
  0x2 u16 flags_2 -- E read with +0 as one u32: bit 0 (0x10000) draws the static quad and UV window instead of bitmap sprites (0x002246e0)
  0x4 u16 vertex_shader -- E 0x00224405: +7 indexes the particle vertex shader table (0x0023ec09); V 0..2
  0x6 unknown 0x2 seen=zero
  0x8 i16 first_sequence -- E 0x00224856: with flag 0x10, sequence = particle +0x16 mod count + first
  0xa i16 sequence_count -- E 0x00224891
  0xc tagref shader_template groups=stem -- V 4,988 of 4,988 resolve to stem
  0x14 block parameters 0x28 align 4 -- V shader parameters by name (docs/formats/effects.md)
    0x0 string name -- V nonzero string IDs
    0x4 unknown 0x4 seen=2 -- 0 or 1
    0x8 tagref bitmap groups=bitm -- V class on all non-null
    0x10 f32 value
    0x14 f32[3] color
    0x20 unknown 0x8 seen=zero -- an empty block descriptor on every tag
  0x1c struct color property -- E 0x002245ad: the RGB colour (0x00246d80), times the emitter tint, clamped
  0x2c struct alpha property -- E 0x002245bb: times the emitter alpha, clamped
  0x3c struct size property -- E 0x00224ca7: times the emitter size
  0x4c struct rotation property -- E 0x00224595: turns, times 2 pi at 0x002245a0
  0x5c struct frame property -- E 0x002248b6: 0..1 over the sequence's sprites, the fraction blending to the next
  0x6c tagref collision_effect groups=effe|foot|snd! -- E pmov kind 1 bounce (0x002b953b) plays it once per particle
  0x74 tagref death_effect groups=effe|foot|snd! -- E 0x00248340 plays it when the particle is freed
  0x7c block block_7c 0x4 align 4
    0x0 string string_0
  0x84 block particle_systems 0x38 align 4 -- V the effe particle-system record (defs/effe.txt): a prt3 reference and emitters
    0x0 tagref particle groups=prt3|PRTM -- E 0x00173cb0 accepts only prt3 and PRTM
    0x8 i32 location -- E read as 32 bits; -1 none
    0xc u16 unknown_c -- E +0xc != 0 is passed as a flag
    0xe u16 unknown_e -- E passed to a compatibility check
    0x10 u16 unknown_10
    0x12 u16 unknown_12 -- E passed to the location lookup
    0x14 i16 unknown_14
    0x16 unknown 0x2 seen=many -- nonzero on 347 instances
    0x18 f32[3] fade -- V (a, b, 1/b) on every instance
    0x24 f32[3] lod -- V (distance, range, 1/range) on every instance
    0x30 block emitters 0xb8 align 4 -- V tiling
      0x0 tagref movement groups=pmov -- V class
      0x8 struct property_8 property
      0x18 struct lifespan property -- E 0x00249170: the particle lifespan, 1 / max(value, minimum)
      0x28 struct property_28 property
      0x38 struct property_38 property
      0x48 struct property_48 property
      0x58 struct property_58 property
      0x68 raw property_68_inputs -- the property's input and range input (i16 each, E 0x00246cd0); the schema reads this word as raw in prt3
      0x6c u16 property_68_output_modifier -- E 0x00246cd0: 1 add, 2 multiply
      0x6e i16 property_68_modifier_input
      0x70 data property_68_function align 4 -- E 0x0013b390
      0x78 u32 emission_shape -- E 0x00249170 switches on it (jump table 0x00249a74)
      0x7c struct shape_input property -- E 0x00249170
      0x8c struct velocity_input property -- E 0x00249170
      0x9c unknown 0x14 seen=many
      0xb0 unknown 0x4 seen=8 -- small integers
      0xb4 unknown 0x4 seen=many -- small integers
  0x8c block postprocess 0x7c align 4 -- the compiled particle shader, the same layout as shad +0x20 (defs/shad.txt); V count 1
    0x0 datum template groups=stem -- E 0x00016bde path; V equals the root template (shader.rs:381)
    0x4 block bitmaps 0xc align 4 -- bitmap parameters, E 0x00016f60 (shader.rs:307). The Halo 1 converter patches +0 of these (h1convert/src/textures.rs:494)
      0x0 datum bitmap groups=bitm -- E 0x00016f60; null selects the rasterizer default ([0x485a80]+0x64) (shader.rs:310)

    0xc block colors 0x4 align 4 -- E 0x00017170 (shader.rs:384)

    0x14 block vectors 0x10 align 4 -- E 0x00017000 (shader.rs:386)

    0x24 block layers 0x2 align 4

    0x2c block passes 0x4 align 4 -- V elements of 0x4 in prt3 (the schema), where shad's are 0x2

      0x2 unknown 0x2
    0x34 block implementations 0xa align 4 -- per-implementation animation ranges into animations, E 0x00017532 (shader.rs:328)

      0x2 u16 render_state_animations -- E 0x00017b60
      0x4 raw stage_and_pixel_constant_animations -- two u16 ranges (stage states E 0x00018560, pixel constants E 0x000176a0); the schema reads this word as raw in prt3
      0x8 u16 vertex_constant_animations -- E 0x00017960: vertex constants and vertex defaults
    0x3c block properties 0x14 align 4 -- animated properties, evaluated by E 0x0001b230 (shader.rs:344)
      0x0 string input -- E 0x000175d9: null = driven by game time, else an object function name (V shield_glow_source, hud_elite, ...)
      0x4 string input_a -- second object-function selector passed to the object callback (shader.rs:350); V position, shield_vitality, ...
      0x8 f32 input_b -- E 0x0001b230: period in seconds when input is null (shader.rs:352); V mostly 1.0
      0xc data function align 4 -- mapping-function blob, E 0x0013b390 evaluates it; when function[1] & 0xf0 == 0, bytes +4/+8 are the f32 output range (shader.rs:353)
    0x44 block property_refs 0x4 align 4 -- E 0x000175d0 (shader.rs:358)
      0x0 i16 property -- E 0x000175d0: index into properties
      0x2 i16 component -- E 0x00017635
    0x4c block property_ranges 0x2 align 4
      0x0 u16 property_refs -- E 0x0001759d: range into property_refs
    0x54 block animations 0x4 align 4 -- animation bindings, E 0x00017568 (shader.rs:365)
      0x0 u8 target -- texture slot, state table index or register, by the range that selects it
      0x1 u8 param_1 -- kind-specific (render-state animations: 1 = scalar output) (shader.rs:371)
      0x2 u8 param_2 -- kind-specific (colour: channel mask and transform) (shader.rs:373)

    0x5c block parameter_bitmaps 0x4 align 4 -- V count 5 in every tag; entry k (k < 4) belongs to root parameter bitmap k (+0x0, +0x8, +0x2c, +0x34); not dereferenced by the draw path (shader.rs:399)
      0x0 i16 bitmap_index -- V index into bitmaps holding the parameter's datum (1,735 of 1,736 for base_map), -1 when the parameter is null; entry 4 has no root parameter
      0x2 unknown 0x2 seen=many -- -1, else 0 or 1
    0x64 block parameter_colors 0xc align 4 -- V count 2; not dereferenced by the draw path
      0x0 f32[3] color -- V equal to root parameter colours +0x10 and +0x3c, all 2,288 tags
    0x6c block parameter_scalars 0x4 align 4 -- V count 6; not dereferenced by the draw path
