# shad: shader instance. The engine draws from the compiled postprocess (+0x20)
# and the root words +0x3e..+0x4c only (docs/formats/shader-instances.md, "shad").
# The parameter block (+0x0c) is never read by the draw path; its values are
# mirrored into postprocess +0x5c/+0x64/+0x6c (V, below) and, through the template,
# into +0x04 bitmaps and +0x0c colours. A tag written from scratch must fill the
# postprocess: that is what the engine binds.
group shad 0x54 align 4
  0x0 tagref template groups=stem -- shader template; E 0x0001663a reads the datum at +4 (shader.rs:282); V always a stem
  0x8 string material_name -- V string IDs read as physical material names (hard_metal_thick, hard_terrain_concrete, brittle_elec_for, ...), null in 924 of 2,288 distinct tags; not read by the draw path
  0xc block parameters 0x50 align 4 -- V count 1 in every tag; not dereferenced by the draw path (shader.rs:284). Mirrored into postprocess +0x5c (bitmaps), +0x64 (colours), +0x6c (scalars)
    0x0 tagref base_map groups=bitm -- V mirrored by postprocess +0x5c[0] (that entry indexes the +0x04 bitmap with this datum in 1,735 of 1,736 set); V names are base/diffuse art (*_dif, gray_50_percent, color_black)
    0x8 tagref self_illum_map groups=bitm -- V mirrored by postprocess +0x5c[1]; V names are *_illum, *_emissive, light_*, default_additive
    0x10 f32[3] color_10 -- V equals postprocess +0x64[0] in all 2,288 tags; nonzero with self_illum_map set in 239, without in 93; role unproven
    0x1c f32 scalar_1c -- V equals postprocess +0x6c[0] when nonzero; nonzero mostly with self_illum_map (198 of 218)
    0x20 f32 scalar_20 -- V postprocess +0x6c[1] slot; zero in every retail tag
    0x24 f32 scalar_24 -- V equals postprocess +0x6c[2] (3 nonzero tags)
    0x28 f32 scalar_28 -- V postprocess +0x6c[3] slot; zero in every retail tag
    0x2c tagref bitmap_2c groups=bitm -- V mirrored by postprocess +0x5c[2]; names mix color_white, alpha_black, foliage and grate maps; role unproven
    0x34 tagref specular_map groups=bitm -- V mirrored by postprocess +0x5c[3]; names *_spec, *_specular, glass_*, color_white; set in 115 tags, exactly those with specular_color nonzero (114 of 115)
    0x3c f32[3] specular_color -- V equals postprocess +0x64[1] in all 2,288 tags; nonzero only with specular_map set (114 of 115)
    0x48 f32 scalar_48 -- V equals postprocess +0x6c[4]; nonzero only with specular_map set (36)
    0x4c f32 scalar_4c -- V equals postprocess +0x6c[5]; nonzero only with bitmap_2c set (13)
  0x14 unknown 0x2 seen=2 default=0100 -- 1 or 5 (5 only with one postprocess LOD)
  0x16 unknown 0x2 seen=4 -- 0..4, mostly 0
  0x18 pad 0x8

    0x0 datum template groups=stem -- E 0x00016bde path; V equals the root template (shader.rs:381)
    0x4 block bitmaps 0xc align 4 -- bitmap parameters, E 0x00016f60 (shader.rs:307). The Halo 1 converter patches +0 of these (h1convert/src/textures.rs:494)

      0x4 i32 image -- E 0x00016f60 to the slot frame; 0x0001b0a0 wraps it by the bitmap's image count (shader.rs:312); V 0 in all but 5 elements
      0x8 f32 lod_offset -- E 0x0001b15c adds it to the draw's LOD term before mip selection (shader.rs:315)
    0xc block colors 0x4 align 4 -- E 0x00017170 (shader.rs:384)

    0x14 block vectors 0x10 align 4 -- E 0x00017000 (shader.rs:386)

    0x1c block lods 0x6 align 4 -- E 0x000174d0 (shader.rs:319)

      0xc data function align 4 -- mapping-function blob, E 0x0013b390 evaluates it; when function[1] & 0xf0 == 0, bytes +4/+8 are the f32 output range (shader.rs:353)
    0x44 block property_refs 0x4 align 4 -- E 0x000175d0 (shader.rs:358)

      0x2 i16 component -- E 0x00017635
    0x4c block property_ranges 0x2 align 4

    0x54 block animations 0x4 align 4 -- animation bindings, E 0x00017568 (shader.rs:365)

      0x1 u8 param_1 -- kind-specific (render-state animations: 1 = scalar output) (shader.rs:371)
      0x2 u8 param_2 -- kind-specific (colour: channel mask and transform) (shader.rs:373)

    0x5c block parameter_bitmaps 0x4 align 4 -- V count 5 in every tag; entry k (k < 4) belongs to root parameter bitmap k (+0x0, +0x8, +0x2c, +0x34); not dereferenced by the draw path (shader.rs:399)
      0x0 i16 bitmap_index -- V index into bitmaps holding the parameter's datum (1,735 of 1,736 for base_map), -1 when the parameter is null; entry 4 has no root parameter
      0x2 unknown 0x2 seen=many -- -1, else 0 or 1
    0x64 block parameter_colors 0xc align 4 -- V count 2; not dereferenced by the draw path
      0x0 f32[3] color -- V equal to root parameter colours +0x10 and +0x3c, all 2,288 tags
    0x6c block parameter_scalars 0x4 align 4 -- V count 6; not dereferenced by the draw path
