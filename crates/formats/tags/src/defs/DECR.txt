# Decorator sets share their indexed mesh records with model particles.
# Block strides and relocation fields follow schema/layouts.txt.

struct indexed_mesh_part 0x8
  0x0 string model_name doc="Identifies the mesh that uses this slice of the index list."
  0x4 i16 first_index doc="The mesh starts at this entry in the index list."
  0x6 i16 index_count doc="Number of index entries assigned to this mesh."

struct decorator_mesh_vertex 0x38
  0x0 f32[3] position doc="Coordinates of the vertex in the mesh."
  0xc f32[3] normal doc="Surface direction perpendicular to the vertex's tangent plane."
  0x18 f32[3] tangent doc="First tangent direction at the vertex."
  0x24 f32[3] bitangent doc="Second tangent direction at the vertex."
  0x30 f32[2] uv doc="Texture coordinates for this vertex."

group DECR 0x6c align 4
  0x0 block shader_palette 0x8 align 4 doc="Shaders available to the decorator variants."
    0x0 tagref shader doc="Selects the shader for this palette entry."
  0x8 f32[2] light_scale_bounds doc="Lower and upper light multipliers for this set."
  0x10 block decorator_classes 0x14 align 4 doc="Groups the available variants by decorator class."
    0x0 string class_name doc="Identifies this class within the set."
    0x4 enum8 decorator_kind doc="Selects the rendering type for this class."
    0x5 unknown 0x3
    0x8 f32 class_scale doc="Overall size multiplier for decorators of this class."
    0xc block variants 0x28 align 4 doc="Alternative appearances within this class."
      0x0 string variant_name doc="Identifies the variant."
      0x4 i8 shader_slot doc="Index of this variant's shader in the set's palette."
      0x5 unknown 0x3
      0x8 flags8 variant_flags doc="Rendering switches for this variant."
      0x9 enum8 fade_distance_class doc="Selects the distance category for fading."
      0xa i8 variant_index doc="Index stored with this variant."
      0xb i8 selection_weight doc="Weight of the variant when distributing decorators."
      0xc f32[2] size_bounds doc="Range of scale factors available to the variant."
      0x14 u32 primary_tint doc="First packed tint colour for the variant."
      0x18 u32 secondary_tint doc="Second packed tint colour for the variant."
      0x1c f32 texture_tint_weight doc="Contribution of the base texture to the tint."
      0x20 f32 lightmap_tint_weight doc="Contribution of the lightmap to the tint."
      0x24 f32 wind_multiplier doc="Scales the effect of wind on the variant."
  0x18 block meshes 0x8 align 4 doc="Index ranges for the named decorator meshes."
    0x0 struct mesh indexed_mesh_part doc="Names the mesh and locates its indices."
  0x20 block vertices 0x38 align 4 doc="Vertex attributes for the decorator meshes."
    0x0 struct vertex decorator_mesh_vertex doc="Geometry and texture coordinates at one vertex."
  0x28 block indices 0x2 align 4 doc="Vertex indices shared by the meshes."
    0x0 i16 vertex_index doc="Selects a vertex for this entry in the mesh."
  0x30 unknown 0x8
  0x38 raw block_raw doc="Locates the external geometry block in its resource file."
  0x3c rawsize block_size of=0x38 doc="Length in bytes of the external geometry block."
  0x40 u32 section_struct_size doc="Length in bytes of the section records in the geometry block."
  0x44 u32 resource_area_size doc="Length in bytes of the geometry block's resource area."
  0x48 block resources 0x10 align 4 doc="Describes the sections of the external geometry data."
    0x0 struct resource block_resource doc="Locates one resource within the geometry block."
  0x50 datum owner doc="Tag that owns the geometry block."
  0x54 i16 runtime_block_offset doc="Relative location of the geometry section for this block."
  0x56 unknown 0x16
