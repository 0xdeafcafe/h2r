# Point physics. Shared contrail and antenna tags on the retail discs pair
# mass, density and drag at the same offsets. H1's precomputed buoyancy
# distinguishes air and water. The H2 root has no child blocks or resources.
group pphy 0x40 align 4
  0x0 u32 flags
  0x4 f32 mass
  0x8 f32 liquid_buoyancy
  0xc f32 air_buoyancy
  0x10 pad 0x10
  0x20 f32 density
  0x24 f32 air_drag
  0x28 f32 liquid_drag
  0x2c f32 friction
  0x30 f32 restitution
  0x34 pad 0xc
