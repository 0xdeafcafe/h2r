# phmo: physics model. Names from crates/formats/tags/src/collision.rs (PhysicsModel::decode) and
# docs/formats/collision.md ("phmo root", "Shapes", "Rigid bodies"). Strides E 0x001ea9c0 / 0x001eaf30.
# Shape references are {u16 type, u16 index}; types (E 0x001eaf30): 0 sphere, 1 pill, 2 box,
# 3 triangle, 4 polyhedron, 5 (0xb0 block), 6 (0x20 block), 0xe list, 0xf mopp.

# Common 0x20-byte header of spheres, pills, boxes, triangles and polyhedra (collision.rs ShapeHeader).
struct shape_header 0x20
  0x0 string name -- collision.rs ShapeHeader::name
  0x4 u16 material -- V index into the root materials block (+0x40) on every retail shape (collision.md "Shapes")
  0x6 unknown 0x2 seen=many
  0x8 unknown 0x4 seen=const -- floats
  0xc unknown 0x4 seen=many -- floats
  0x10 unknown 0x4 seen=many -- floats
  0x14 unknown 0x4 seen=many -- floats
  0x18 unknown 0x4 seen=many -- floats
  0x1c unknown 0x4 seen=many

# A constraint's pivot frame in one body's node space (D, retail ragdolls: three unit axes and a
# position; the hinge's frame B and every ragdoll frame A sit on the node's origin).
struct constraint_frame 0x34
  0x0 unknown 0x4 seen=const -- floats; 1.0 on every retail constraint (meaning not established)
  0x4 f32[3] forward -- D unit axis
  0x10 f32[3] left -- D unit axis
  0x1c f32[3] up -- D unit axis
  0x28 f32[3] position -- D the pivot, in the node's space

# The 0x78 bytes every constraint element starts with (E 0x001d77d0 reads +4/+6 as the two nodes and
# +0x70 as the edge; collision.md "Constraints").
struct constraint_header 0x78
  0x0 string name
  0x4 i16 node_a -- E 0x001d786f: -1 or a phmo node; mapped to its body, both must differ
  0x6 i16 node_b -- E 0x001d7895: as node_a
  0x8 struct frame_a constraint_frame -- D: in node_a's space
  0x3c struct frame_b constraint_frame -- D: in node_b's space
  0x70 i16 edge -- E 0x001d78fb: index into node_edges, whose constraint list gives the friction
  0x72 pad 0x2
  0x74 unknown 0x4 seen=zero
